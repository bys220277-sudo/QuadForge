cmake_minimum_required(VERSION 3.16)
project(quadforge_native LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)

file(GLOB QF_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(quadforge_native SHARED ${QF_SOURCES})
target_include_directories(quadforge_native PRIVATE src)
set_target_properties(quadforge_native PROPERTIES OUTPUT_NAME "quadforge_native")
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(quadforge_native PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(quadforge_native PRIVATE -O3 -Wall -Wextra -Wno-unused-parameter -fno-math-errno)
endif()
if(MSVC)
  target_compile_options(quadforge_native PRIVATE /O2 /fp:precise /EHsc /bigobj)
endif()
